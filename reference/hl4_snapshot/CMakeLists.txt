cmake_minimum_required(VERSION 3.22)
project(heroes_lore_4 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HL4_SWITCH "Build libnx homebrew" OFF)
option(HL4_TCI "Use pinned experimental Unicorn interpreter" OFF)
set(UNICORN_ARCH arm CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_FORCE_STATIC_VCRT ON CACHE BOOL "" FORCE)
if(HL4_TCI)
  set(UNICORN_INTERPRETER ON CACHE BOOL "" FORCE)
  set(UC_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/unicorn-tci/unicorn-engine-sys-tci")
else()
  set(UC_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/unicorn")
endif()
add_subdirectory("${UC_SOURCE}" unicorn-build)
if(HL4_SWITCH)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  set(SDL_TARGET PkgConfig::SDL2)
elseif(ANDROID)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/SDL2 SDL2-build)
  set(SDL_TARGET SDL2)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/SDL2 SDL2-build)
  set(SDL_TARGET SDL2-static)
endif()
if(ANDROID)
  add_library(main SHARED src/main.cpp)
  set(HL4_TARGET main)
else()
  add_executable(heroes_lore_4 src/main.cpp)
  set(HL4_TARGET heroes_lore_4)
endif()
target_include_directories(${HL4_TARGET} PRIVATE "${UC_SOURCE}/include")
target_link_libraries(${HL4_TARGET} PRIVATE unicorn ${SDL_TARGET})
if(HL4_TCI)
  target_compile_definitions(${HL4_TARGET} PRIVATE HL4_TCI=1)
endif()
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/runtime/hl4-icon.ico")
  enable_language(RC)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hl4-icon.rc" "1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/build/runtime/hl4-icon.ico\"\n")
  target_sources(${HL4_TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/hl4-icon.rc")
endif()
if(MSVC)
  target_compile_options(${HL4_TARGET} PRIVATE /O2 /utf-8 /D_CRT_SECURE_NO_WARNINGS)
else()
  if(HL4_SWITCH)
    target_compile_options(${HL4_TARGET} PRIVATE -O3)
  else()
    target_compile_options(${HL4_TARGET} PRIVATE -O2)
  endif()
endif()
if(MINGW)
  target_link_options(${HL4_TARGET} PRIVATE -static-libgcc -static-libstdc++)
endif()
if(ANDROID)
  target_link_libraries(main PRIVATE log android)
elseif(HL4_SWITCH)
  target_sources(heroes_lore_4 PRIVATE platforms/switch/unicorn_os.c)
  target_link_libraries(heroes_lore_4 PRIVATE EGL glapi drm_nouveau nx)
endif()
